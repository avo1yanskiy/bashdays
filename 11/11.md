```Dockerfile
FROM ubuntu:22.04

ENV DEBIAN_FRONTEND=noninteractive

RUN apt-get update && \
    apt-get install -y \
    docker.io \
    iptables \
    ca-certificates \
    curl \
    fuse-overlayfs \
    && rm -rf /var/lib/apt/lists/*

RUN echo '{"hosts": ["unix:///var/run/docker.sock"], "storage-driver": "vfs"}' > /etc/docker/daemon.json

CMD ["dockerd"]
```
